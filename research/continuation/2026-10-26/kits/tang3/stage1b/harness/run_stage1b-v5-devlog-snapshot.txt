def _load(name: str, filename: str):
    if name in sys.modules:
        return sys.modules[name]
    spec = importlib.util.spec_from_file_location(name, HERE / filename)
    module = importlib.util.module_from_spec(spec)
    sys.modules[name] = module
    spec.loader.exec_module(module)
    return module

def sha256_file(path) -> str:
    digest = hashlib.sha256()
    with open(path, "rb") as handle:
        for chunk in iter(lambda: handle.read(1 << 20), b""):
            digest.update(chunk)
    return digest.hexdigest()

def runner_problems() -> tuple[list[str], dict]:
    if not RUNNER_MANIFEST.exists():
        return [f"refused: {RUNNER_MANIFEST.name} does not exist (publish the Runner and write its manifest first)"], {}
    identity = ce.runner_identity_from_disk(RUNNER, RUNNER_MANIFEST, "runner-tier3-v3")
    problems = list(identity["problems"])
    if identity.get("dll") == ce.OLD_RUNNER_DLL or identity.get("tree") == ce.OLD_RUNNER_TREE:
        problems.append("runner-tier3-v3 has the archived v2 DLL or tree; it is not the cost-first build")
    if RUNNER.resolve() == pathlib.Path(OLD_RUNNER).resolve():
        problems.append("RUNNER points at runner-tier3-v2")
    if not PROVENANCE.exists():
        problems.append(f"{PROVENANCE.name} (filled runner provenance receipt) does not exist")
    else:
        text = PROVENANCE.read_text(encoding="utf-8")
        if identity.get("ok") and (identity["tree"] not in text or identity["dll"] not in text):
            problems.append(f"{PROVENANCE.name} does not record this tree hash and DLL SHA-256")
        unfilled = re.findall(r"`<[^`\n]*>`", text)
        if unfilled:
            problems.append(f"{PROVENANCE.name} still has {len(unfilled)} unfilled template field(s), e.g. {unfilled[0]}")
    return problems, identity


def config_problems() -> list[str]:
    problems = []
    for arm in ARMS:
        path = run_pilot.CONFIGS / f"{arm}.json"
        if not path.exists() or sha256_file(path) != run_t38.CONFIG_SHA256[arm]:
            problems.append(f"commitment config {arm} missing or SHA-256 mismatch")
    for mode, path in (("off", WP4_OFF), ("on", WP4_ON)):
        if not path.exists():
            problems.append(f"WP4 {mode} missing: {path}")
            continue
        raw = path.read_bytes()
        if hashlib.sha256(raw).hexdigest() != WP4_SHA256[mode]:
            problems.append(f"WP4 {mode} file SHA-256 mismatch")
        try:                                                  # N4: a malformed config is a problem, not an escape
            if cj.canonical_sha256(raw) != WP4_CANONICAL[mode]:
                problems.append(f"WP4 {mode} canonical SHA-256 mismatch")
            for arm in ARMS:
                commitment = run_pilot.CONFIGS / f"{arm}.json"
                if commitment.exists() and cj.binding(commitment.read_bytes(), raw) != ce.BINDINGS[(mode, arm)]:
                    problems.append(f"binding {mode}/{arm} differs from {ce.BINDINGS[(mode, arm)][:8]}...")
        except (cj.CanonicalJsonError, ValueError, UnicodeDecodeError) as error:
            problems.append(f"WP4 {mode} is not valid canonical JSON ({type(error).__name__}: {error})")
    if WP4_OFF.exists() and WP4_ON.exists():
        try:
            on = cj.parse_strict(WP4_ON.read_bytes())
            if on.pop("selectionObjectiveOrder", None) != "cost-first" or \
                    cj.canonical_bytes_of_value(on) != cj.canonical_bytes(WP4_OFF.read_bytes()):
                problems.append("WP4 ON is not exactly v2 + selectionObjectiveOrder=cost-first")
            if "selectionObjectiveOrder" in cj.parse_strict(WP4_OFF.read_bytes()):
                problems.append("WP4 OFF declares selectionObjectiveOrder (it must be absent, R22:50,232)")
        except (cj.CanonicalJsonError, ValueError, UnicodeDecodeError, AttributeError) as error:
            problems.append(f"WP4 ON/OFF comparison impossible ({type(error).__name__}: {error})")
    return problems

    return problems

def tree_files(root: pathlib.Path, excludes=()) -> list[pathlib.Path]:
    out = []
    for directory, dirnames, filenames in os.walk(root):
        dirnames.sort()
        for filename in filenames:
            path = pathlib.Path(directory) / filename
            relative = path.relative_to(root).as_posix()
            if not any(relative.startswith(prefix) for prefix in excludes):
                out.append(path)
    return sorted(out, key=lambda p: p.relative_to(root).as_posix())


def tree_body(root: pathlib.Path, excludes=()) -> list[str]:
    """runner_manifest.py:17-23 line format (relative path, SHA-256, bytes), sorted, minus excluded prefixes."""
    return [f"{p.relative_to(root).as_posix()}\t{sha256_file(p)}\t{p.stat().st_size}"
            for p in tree_files(root, excludes)]

def process_resolve(name: str, *, environ=None, app_dir=None, cwd=None, windows=None, is_file=None) -> str | None:
    """v4 (R38 s6): the executable that subprocess (shell=False, Windows) starts for a bare command name. CreateProcess
    appends '.exe' when the name has no extension and searches, in order: 1 the directory of the calling executable,
    2 the caller's current directory, 3 the 32-bit system directory, 4 the 16-bit system directory, 5 the Windows
    directory, 6 the PATH entries in order. The current directory is always searched here (conservative: the frozen
    switch NoDefaultCurrentDirectoryInExePath could only remove it). PATHEXT and App Paths are not used."""
    environ = os.environ if environ is None else environ
    upper = {key.upper(): value for key, value in environ.items()}
    is_file = os.path.isfile if is_file is None else is_file
    filename = name if pathlib.PurePath(name).suffix else name + ".exe"
    windows = windows or upper.get("SYSTEMROOT") or r"C:\Windows"
    directories = [app_dir, cwd, os.path.join(windows, "System32"), os.path.join(windows, "System"), windows]
    directories += [entry.strip().strip('"') for entry in upper.get("PATH", "").split(";")]
    for directory in directories:
        if directory:
            candidate = os.path.join(str(directory), filename)
            if is_file(candidate):
                return pathlib.Path(candidate).as_posix()
    return None


def tool_resolution(name: str, context: str, environ=None) -> str | None:
    """The tool a bare name reaches in the job context (job interpreter directory, job cwd) or the driver context
    (this interpreter's directory, this process's cwd)."""
    if context == "job":
        return process_resolve(name, environ=environ, app_dir=JOB_APP_DIR, cwd=JOB_CWD)
    return process_resolve(name, environ=environ, app_dir=pathlib.Path(sys.executable).parent, cwd=os.getcwd())


def resolve_tool(name: str, environ=None, which=None) -> str | None:
    """v3 name kept: the job-context resolution (v4 no longer uses shutil.which, which ignores CreateProcess's order)."""
    return tool_resolution(name, "job", environ)


def pth_lines(site: pathlib.Path = None) -> list[tuple[str, str]]:
    """Every non-comment line of the .pth files of one site directory (site.py processes them at start-up)."""
    site = PYTHON_SITE if site is None else site
    out = []
    for pth in sorted(pathlib.Path(site).glob("*.pth")):
        for line in pth.read_text(encoding="utf-8", errors="replace").splitlines():
            if line.strip() and not line.startswith("#"):
                out.append((pth.name, line))
    return out


def _resolved(path) -> str:
    """v4 (R38-5): normalized and resolved ('..', junctions, symbolic links), case-folded, forward slashes."""
    return os.path.normcase(os.path.realpath(os.path.normpath(str(path)))).replace("\\", "/").rstrip("/")


def _inside(path: str, roots) -> bool:
    """v4 (R38-5): containment AFTER resolution of the candidate and every root (v3 compared lexical paths)."""
    candidate = _resolved(path)
    for root in roots:
        base = _resolved(root)
        if candidate == base or candidate.startswith(base + "/"):
            return True
    return False


def pth_redirect_problems(lines, site, roots) -> list[str]:
    """N3 + R38-5: a .pth path line adds sys.path entry os.path.join(site, line) (site.makepath); after resolution it
    must stay inside the frozen trees. An import line executes code: it must be a reviewed line (name + SHA-256)."""
    problems = []
    for name, line in lines:
        if line.startswith(("import ", "import\t")):
            digest = hashlib.sha256(line.encode("utf-8")).hexdigest()
            if (name, digest) not in REVIEWED_PTH_IMPORTS:
                problems.append(f"{name}: executable import line {line[:60]!r} (sha256 {digest[:12]}...) is not a "
                                f"reviewed line")
            continue
        target = os.path.join(str(site), line.rstrip())
        if not _inside(target, roots):
            problems.append(f"{name}: path entry {line.strip()!r} resolves to {_resolved(target)}, outside the frozen "
                            f"trees")
    return problems


def pth_audit_problems(roots) -> list[str]:
    """Both site directories the job interpreter processes."""
    problems = []
    for directory in PTH_DIRS:
        problems += pth_redirect_problems(pth_lines(directory), directory, roots)
    return problems


FILE_LINE = re.compile(r"^([0-9a-f]{64})  (\S.*)$")
TREE_LINE = re.compile(r"^tree (\S+) ([0-9a-f]{64}) (\d+) (\S.*)$")
ENV_LINE = re.compile(r"^env ([^=\s]+)=(.*)$")
ABSENT_LINE = re.compile(r"^absent (\S.*)$")
TOOL_LINE = re.compile(r"^tool (\S+) ([0-9a-f]{64}) (\S.*)$")      # v4: name, SHA-256, resolved path


def freeze_problems(text: str | None = None, *, targets=None, trees=None, environ=None, absent=None,
                    sha=sha256_file, exists=lambda p: pathlib.Path(p).exists(), read=None, body=None,
                    resolve=None, pth=None, site=None) -> list[str]:
    """F2 + R28 N3/N4 + R38: exact verification of the freeze. Every dependency is injectable so the self-test runs in
    memory; an unreadable input is a problem, never an exception. resolve(name, context) is the CreateProcess
    resolution in the 'job' and 'driver' contexts (default tool_resolution)."""
    if text is None:
        if not FREEZE.exists():
            return [f"{FREEZE.name} does not exist (run `freeze` after approval, before sealing)"]
        text = FREEZE.read_text(encoding="utf-8")
    targets = freeze_targets() if targets is None else targets
    trees = EXTERNAL_TREES if trees is None else trees
    absent = ABSENT_REQUIRED if absent is None else absent
    read = (lambda p: pathlib.Path(p).read_text(encoding="utf-8")) if read is None else read
    body = tree_body if body is None else body
    resolve = (lambda name, context: tool_resolution(name, context, environ)) if resolve is None else resolve
    problems, files, tree_lines, env_lines, absent_lines, tool_lines = [], {}, {}, {}, set(), {}
    for number, line in enumerate(text.splitlines(), start=1):
        if not line or line.startswith("#"):
            continue
        match = FILE_LINE.match(line)
        if match:
            if match.group(2) in files:
                problems.append(f"freeze line {number}: duplicate file entry {match.group(2)}")
            files[match.group(2)] = match.group(1)
            continue
        match = TREE_LINE.match(line)
        if match:
            if match.group(1) in tree_lines:
                problems.append(f"freeze line {number}: duplicate tree entry {match.group(1)}")
            tree_lines[match.group(1)] = (match.group(2), int(match.group(3)), match.group(4))
            continue
        match = ENV_LINE.match(line)
        if match:
            if match.group(1) in env_lines:
                problems.append(f"freeze line {number}: duplicate env entry {match.group(1)}")
            env_lines[match.group(1)] = match.group(2)
            continue
        match = ABSENT_LINE.match(line)
        if match:
            if match.group(1) in absent_lines:
                problems.append(f"freeze line {number}: duplicate absent entry {match.group(1)}")
            absent_lines.add(match.group(1))
            continue
        match = TOOL_LINE.match(line)
        if match:
            if match.group(1) in tool_lines:
                problems.append(f"freeze line {number}: duplicate tool entry {match.group(1)}")
            tool_lines[match.group(1)] = (match.group(2), match.group(3))
            continue
        problems.append(f"freeze line {number}: malformed: {line[:80]!r}")
    expected = {pathlib.Path(p).as_posix() for p in targets}
    if not expected or not files:
        problems.append("freeze lists no file or the required file inventory is empty")
    missing, extra = sorted(expected - set(files)), sorted(set(files) - expected)
    if missing:
        problems.append(f"freeze does not cover {len(missing)} required file(s), e.g. {missing[:3]}")
    if extra:
        problems.append(f"freeze lists {len(extra)} file(s) outside the required inventory, e.g. {extra[:3]}")
    for path in sorted(expected & set(files)):
        try:
            if not exists(path) or sha(path) != files[path]:
                problems.append(f"frozen input changed or missing: {path}")
        except (OSError, ValueError) as error:
            problems.append(f"frozen input unreadable: {path} ({type(error).__name__}: {error})")
    want_trees = {tree_id: (pathlib.Path(root).as_posix(), excludes) for tree_id, root, excludes in trees}
    if set(tree_lines) != set(want_trees):
        problems.append(f"freeze trees {sorted(tree_lines)} != required external trees {sorted(want_trees)}")
    for tree_id in sorted(set(tree_lines) & set(want_trees)):
        tree_hash, count, root = tree_lines[tree_id]
        want_root, excludes = want_trees[tree_id]
        if root != want_root:
            problems.append(f"external tree {tree_id}: root {root} != {want_root}")
            continue
        manifest = external_manifest(tree_id)
        try:
            manifest_text = read(manifest)
        except (OSError, ValueError, KeyError):
            problems.append(f"external tree {tree_id}: manifest {manifest.name} unreadable")
            continue
        if manifest_text.split("\n", 1)[0] != f"# tree-sha256 {tree_hash} files {count}":
            problems.append(f"external tree {tree_id}: manifest header differs from the freeze line")
        if not exists(root):
            problems.append(f"external tree {tree_id}: root {root} missing")
            continue
        try:
            live = body(pathlib.Path(root), excludes)
        except (OSError, ValueError) as error:
            problems.append(f"external tree {tree_id}: unreadable ({type(error).__name__}: {error})")
            continue
        if ce.tree_text(live) != manifest_text:
            frozen = manifest_text.split("\n")[1:-1]
            changed = sorted(set(frozen) ^ set(live))
            problems.append(f"external tree {tree_id} differs from its frozen manifest ({len(changed)} differing "
                            f"line(s), e.g. {[c.split(chr(9))[0] for c in changed[:3]]})")
    current = frozen_env(environ)
    for name in sorted(set(current) | set(env_lines)):
        if name not in env_lines:
            problems.append(f"environment variable {name} is set now but was not set at freeze time")
        elif name not in current:
            problems.append(f"environment variable {name} was frozen but is not set now")
        elif current[name] != env_lines[name]:
            problems.append(f"environment variable {name} changed since the freeze")
    problems += forbidden_env(environ)
    if set(tool_lines) != set(REQUIRED_TOOLS):
        problems.append(f"freeze tools {sorted(tool_lines)} != required resolved tools {list(REQUIRED_TOOLS)}")
    roots = [pathlib.Path(root).as_posix() for _, root, _ in trees]
    problems += tool_problems(tool_lines, roots, resolve, sha)
    try:                                                       # R38-5: resolved paths, reviewed import lines
        problems += (pth_redirect_problems(pth(), site, roots) if pth is not None else pth_audit_problems(roots))
    except (OSError, ValueError) as error:
        problems.append(f".pth audit impossible ({type(error).__name__}: {error})")
    want_absent = {pathlib.Path(p).as_posix() for p in absent}
    if absent_lines != want_absent:
        problems.append(f"freeze absent paths {sorted(absent_lines)} != required {sorted(want_absent)}")
    for path in sorted(want_absent):
        if exists(path):
            problems.append(f"required-absent path now exists (would change imports): {path}")
    return problems


def tool_problems(tool_lines: dict, roots, resolve, sha) -> list[str]:
    """R38 s6 + R41-4: each required tool resolves, in the job and the driver context, to the frozen path with the
    frozen SHA-256; it lies inside the frozen trees, or at the exact path of its declared TOOL_EXCEPTIONS entry."""
    problems = []
    for name in sorted(set(tool_lines) & set(REQUIRED_TOOLS)):
        frozen_sha, frozen_path = tool_lines[name]
        try:
            job, driver = resolve(name, "job"), resolve(name, "driver")
        except Exception as error:
            problems.append(f"tool {name} cannot be resolved ({type(error).__name__}: {error})")
            continue
        if job != frozen_path:
            problems.append(f"tool {name} now resolves to {job} in the job context, frozen {frozen_path}")
            continue
        if driver != job:
            problems.append(f"tool {name} resolves to {driver} for the driver but {job} for the jobs")
        try:
            if sha(job) != frozen_sha:
                problems.append(f"tool {name} executable bytes changed since the freeze: {job}")
        except Exception as error:
            problems.append(f"tool {name} executable unreadable: {job} ({type(error).__name__}: {error})")
        problems += tool_location_problems(name, job, roots)
    return problems


def tool_location_problems(name: str, path: str | None, roots) -> list[str]:
    """R41-4: a tool lies inside the frozen trees, or at exactly the path its declared exception admits."""
    if path is None:
        return [f"tool {name} does not resolve"]
    exception = TOOL_EXCEPTIONS.get(name)
    if exception is None:
        return [] if _inside(path, roots) else [f"tool {name} resolves outside the frozen trees: {path}"]
    if _resolved(path) != _resolved(exception["path"]):
        return [f"tool {name} resolves to {path}; its declared exception admits only {exception['path']}"]
    return []


def seal_problems() -> list[str]:
    if not SEAL.exists():
        return [f"refused: {SEAL.name} does not exist (the plan is not sealed)"]
    problems, listed = [], set()
    for line in SEAL.read_text(encoding="utf-8").splitlines():
        if not line.strip():
            continue
        match = re.match(r"^([0-9a-f]{64}) [ *]?(\S+)", line)
        if not match:
            problems.append(f"seal line malformed: {line[:60]}")
            continue
        digest, name = match.group(1), match.group(2).lstrip("*")
        if name in listed:
            problems.append(f"seal lists {name} twice")
        listed.add(name)
        target = HERE / name
        if not target.exists() or sha256_file(target) != digest:
            problems.append(f"sealed file changed or missing: {name}")
    missing = [name for name in SEALED_REQUIRED if name not in listed]
    if missing:
        problems.append(f"seal does not cover {missing}")
    return problems


def latch_problems(phase: str, for_comparator: bool = False) -> list[str]:
    """F3 + R28 N3: any failure verdict or the input-drift stop receipt stops every job-start path and every verdict;
    a phase with its own pass verdict is closed. Used by `run`, by every job right before launch and by the
    comparator."""
    problems = [f"refused: {failed.name} exists; Stage 1b stops (R22:208, R17 G0)"
                for failed in (GATE0_FAILED, SMOKE_FAILED) if failed.exists()]
    if DRIFT_STOP.exists():
        problems.append(f"refused: {DRIFT_STOP.name} exists; frozen inputs drifted, Stage 1b stops (R28 N3)")
    own = {"gate0": GATE0_PASSED, "smoke": SMOKE_PASSED}.get(phase)
    if own is not None and own.exists():
        problems.append(f"refused: {own.name} exists; phase {phase} is closed")
    return problems


def verdict_problems(path, failed, word: str, identity: dict) -> list[str]:
    if failed.exists():
        return [f"refused: {failed.name} exists; Stage 1b stops (R22:208, R17 G0)"]
    if not path.exists():
        return [f"refused: {path.name} does not exist"]
    text = path.read_text(encoding="utf-8").splitlines()
    problems = []
    if not text or text[-1] != f"{word} PASSED":
        problems.append(f"{path.name} does not end with '{word} PASSED'")
    if not identity.get("ok") or f"runnerDllSha256 {identity['dll']}" not in text \
            or f"runnerTreeSha256 {identity['tree']}" not in text:
        problems.append(f"{path.name} was not written for this verified Runner build")
    return problems


def inspection_problems() -> list[str]:
    """F4 (R24:114): main needs the recorded integration inspection of the smoke report: SMOKE-INSPECTION.txt,
    written by the person who read the smoke report, with the lines `smokeVerdictSha256 <SHA-256 of
    SMOKE-PASSED.txt>`, `inspected-by: <names>` and `inspection: accepted`."""
    if not INSPECTION.exists():
        return [f"refused: {INSPECTION.name} does not exist (record the smoke integration inspection first)"]
    lines = INSPECTION.read_text(encoding="utf-8").splitlines()
    problems = []
    if not SMOKE_PASSED.exists() or f"smokeVerdictSha256 {sha256_file(SMOKE_PASSED)}" not in lines:
        problems.append(f"{INSPECTION.name} does not name the SHA-256 of {SMOKE_PASSED.name}")
    if "inspection: accepted" not in lines:
        problems.append(f"{INSPECTION.name} does not record 'inspection: accepted'")
    if not any(line.startswith("inspected-by: ") and line[len("inspected-by: "):].strip() for line in lines):
        problems.append(f"{INSPECTION.name} does not name the inspector ('inspected-by: ...')")
    return problems

def _guard(label: str, check, default=()):
    """N4: an input check that raises becomes an input problem."""
    try:
        return check()
    except Exception as error:
        return list(default) + [f"{label} could not be validated ({type(error).__name__}: {error})"]


def input_integrity_problems(phase: str) -> tuple[list[str], dict]:
    """Every frozen-input check of a phase (no latches, no verdicts); guarded (N4)."""
    cells = sorted({cell for _, _, cell, _, _ in jobs(phase)})
    try:
        runner, identity = runner_problems()
    except Exception as error:
        runner, identity = [f"runner identity could not be validated ({type(error).__name__}: {error})"], {}
    problems = list(runner)
    problems += _guard("WP4/commitment configs", config_problems)
    problems += _guard("harness root", harness_problems)
    problems += _guard("cell inputs", lambda: cell_problems(cells))
    if phase == "gate0":
        problems += _guard("Gate-0 references", reference_problems)
    problems += _guard("plan seal", seal_problems)
    problems += _guard("input freeze", freeze_problems)
    return problems, identity


def gating_problems(phase: str, identity: dict) -> list[str]:
    """Run gates that are not input integrity (latches, AC policy, prior verdicts, smoke inspection). They refuse a
    run; they are never recorded as input drift (R38-2: no invented drift)."""
    problems = latch_problems(phase) + environment_problems()
    if phase in ("smoke", "main"):
        problems += _guard("Gate-0 verdict", lambda: verdict_problems(GATE0_PASSED, GATE0_FAILED, GATE0_WORD,
                                                                      identity))
    if phase == "main":
        problems += _guard("smoke verdict", lambda: verdict_problems(SMOKE_PASSED, SMOKE_FAILED, "SMOKE", identity))
        problems += _guard("smoke inspection", inspection_problems)
    return problems


ANALYSIS_LINE = re.compile(r"^([0-9a-f]{64})  ([A-Za-z0-9][A-Za-z0-9._-]*)$")


def analysis_freeze_problems(text: str | None = None, *, sha=None, exists=None, read=None) -> list[str]:
    """R38 s6 + R41-1 (plan v5 section 12): the main-phase analysis code (CF extraction, reporting, interaction
    identity, guards, cell/day handling, their dependencies) is frozen by a hashed amendment BEFORE any main job
    starts. stage1b-analysis-freeze.sha256 lists '<sha256>  <bare file name in this folder>' lines (sha256sum format),
    no duplicate; every listed file must exist with that SHA-256; the list must include STAGE1B-ANALYSIS-AMENDMENT.md,
    whose 'inventory-reviewed-by: <names>' and 'inventory-coverage: complete' lines record the manual review that the
    list covers the promised extractor, report and dependencies."""
    sha = (lambda name: sha256_file(HERE / name)) if sha is None else sha
    exists = (lambda name: (HERE / name).is_file()) if exists is None else exists
    read = (lambda name: (HERE / name).read_text(encoding="utf-8")) if read is None else read
    if text is None:
        if not ANALYSIS_FREEZE.exists():
            return [f"refused: {ANALYSIS_FREEZE.name} does not exist (freeze the main-phase analysis code by a hashed "
                    f"amendment before any main job starts; plan section 13)"]
        text = ANALYSIS_FREEZE.read_text(encoding="utf-8")
    problems, listed = [], set()
    for number, line in enumerate(text.splitlines(), start=1):
        if not line or line.startswith("#"):
            continue
        match = ANALYSIS_LINE.match(line)
        if not match:
            problems.append(f"{ANALYSIS_FREEZE.name} line {number} malformed: {line[:60]!r}")
            continue
        digest, name = match.groups()
        if name in listed:
            problems.append(f"{ANALYSIS_FREEZE.name} lists {name} twice")
        listed.add(name)
        try:
            if not exists(name) or sha(name) != digest:
                problems.append(f"analysis input changed or missing: {name}")
        except Exception as error:
            problems.append(f"analysis input unreadable: {name} ({type(error).__name__}: {error})")
    if not listed:
        problems.append(f"{ANALYSIS_FREEZE.name} lists no analysis file")
    if ANALYSIS_AMENDMENT not in listed:
        problems.append(f"{ANALYSIS_FREEZE.name} does not list the amendment record {ANALYSIS_AMENDMENT} (the manual "
                        f"inventory review, R41-1)")
    else:
        try:
            lines = read(ANALYSIS_AMENDMENT).splitlines()
            if not any(line.startswith("inventory-reviewed-by: ") and line[len("inventory-reviewed-by: "):].strip()
                       for line in lines) or "inventory-coverage: complete" not in lines:
                problems.append(f"{ANALYSIS_AMENDMENT} does not record 'inventory-reviewed-by: <names>' and "
                                f"'inventory-coverage: complete'")
        except Exception as error:
            problems.append(f"{ANALYSIS_AMENDMENT} unreadable ({type(error).__name__}: {error})")
    return problems


BINDING_SHA_LINE = re.compile(r"^manifest-sha256 ([0-9a-f]{64})$")
ANALYSIS_BINDING_SHA = None        # R41-1: SHA-256 of the binding file, held since the start of this invocation


def analysis_binding_text(manifest_bytes: bytes, stamp: str) -> str:
    """R41-1: the retained identity: the manifest's SHA-256 and its inventory lines, copied verbatim."""
    inventory = [line for line in manifest_bytes.decode("utf-8").splitlines() if line and not line.startswith("#")]
    return "\n".join([f"# stage1b analysis binding, written {stamp} by run_stage1b-v5.py at the first `run main` "
                      f"admission (R41-1); never rewritten", f"manifest-sha256 {hashlib.sha256(manifest_bytes).hexdigest()}"]
                     + inventory) + "\n"


def bind_analysis() -> str:
    """R41-1: write the binding exactly once (mode "x"), before any main job; returns its SHA-256."""
    stamp = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
    text = analysis_binding_text(ANALYSIS_FREEZE.read_bytes(), stamp)
    with open(ANALYSIS_BINDING, "x", encoding="utf-8", newline="\n") as handle:
        handle.write(text)
    return hashlib.sha256(text.encode("utf-8")).hexdigest()


def main_log_problems(binding_sha: str) -> list[str]:
    """R41-1: every existing main job log names the binding it was admitted under (first line)."""
    problems = []
    for name, *_ in jobs("main"):
        log = LOG_ROOT / "main" / f"{name}.log"
        if log.exists():
            first = log.read_text(encoding="utf-8").split("\n", 1)[0]
            if not first.endswith(f" analysis-binding {binding_sha}"):
                problems.append(f"main log {name}.log was not admitted under the current analysis binding")
    return problems


def analysis_binding_problems(*, binding_text=None, manifest_bytes=None, sha=None, exists=None, held=None,
                              logs=None) -> list[str]:
    """R41-1: compare the CURRENT manifest and files with the RETAINED binding (not the manifest with itself), the
    main logs with the binding, and the binding with the value held since this invocation started."""
    sha = (lambda name: sha256_file(HERE / name)) if sha is None else sha
    exists = (lambda name: (HERE / name).is_file()) if exists is None else exists
    binding_text = ANALYSIS_BINDING.read_text(encoding="utf-8") if binding_text is None else binding_text
    if manifest_bytes is None:
        manifest_bytes = ANALYSIS_FREEZE.read_bytes() if ANALYSIS_FREEZE.exists() else None
    held = ANALYSIS_BINDING_SHA if held is None else held
    problems, bound_manifest, inventory = [], None, []
    for line in binding_text.splitlines():
        if not line or line.startswith("#"):
            continue
        match = BINDING_SHA_LINE.match(line)
        if match and bound_manifest is None:
            bound_manifest = match.group(1)
            continue
        match = ANALYSIS_LINE.match(line)
        if not match:
            problems.append(f"{ANALYSIS_BINDING.name} line malformed: {line[:60]!r}")
            continue
        inventory.append(match.groups())
    if bound_manifest is None or not inventory:
        problems.append(f"{ANALYSIS_BINDING.name} holds no manifest SHA-256 or no inventory")
    if manifest_bytes is None:
        problems.append(f"{ANALYSIS_FREEZE.name} is missing after the analysis binding")
    elif hashlib.sha256(manifest_bytes).hexdigest() != bound_manifest:
        problems.append(f"{ANALYSIS_FREEZE.name} differs from the manifest bound at the first main admission")
    for digest, name in inventory:
        try:
            if not exists(name) or sha(name) != digest:
                problems.append(f"analysis input changed or missing since the binding: {name}")
        except Exception as error:
            problems.append(f"analysis input unreadable: {name} ({type(error).__name__}: {error})")
    binding_sha = hashlib.sha256(binding_text.encode("utf-8")).hexdigest()
    if held is not None and binding_sha != held:
        problems.append(f"{ANALYSIS_BINDING.name} changed during this invocation")
    problems += (main_log_problems if logs is None else logs)(binding_sha)
    return problems


def analysis_problems(phase: str) -> list[str]:
    """R41-1: before the binding, the manifest is checked for consistency; after it, against the binding. A main job
    without a binding means the binding was lost after the main phase began."""
    if phase != "main":
        return []
    if ANALYSIS_BINDING.exists():
        return _guard("analysis binding", analysis_binding_problems)
    problems = _guard("analysis freeze", analysis_freeze_problems)
    if phase_started("main"):
        problems.append(f"{ANALYSIS_BINDING.name} is missing although main jobs exist (lost after the main phase began)")
    return problems


def analysis_paths() -> list[pathlib.Path]:
    """R41-1: the manifest, the binding and the bound files (main phase stat snapshot)."""
    paths = [ANALYSIS_FREEZE, ANALYSIS_BINDING]
    try:
        for line in ANALYSIS_BINDING.read_text(encoding="utf-8").splitlines():
            match = ANALYSIS_LINE.match(line)
            if match:
                paths.append(HERE / match.group(2))
    except OSError:
        pass
    return paths


def frozen_input_problems(phase: str, for_comparator: bool = False) -> list[str]:
    problems, identity = input_integrity_problems(phase)
    if not for_comparator:
        problems += gating_problems(phase, identity) + analysis_problems(phase)
    return problems


def phase_started(phase: str) -> list[str]:
    """R38-2: evidence that a phase has begun: a job output directory or a job log."""
    evidence = []
    for name, *_ in jobs(phase):
        if (OUT_ROOT / phase / name).exists():
            evidence.append(f"output {phase}/{name}")
        if (LOG_ROOT / phase / f"{name}.log").exists():
            evidence.append(f"log {phase}/{name}.log")
    return evidence


def study_started() -> list[str]:
    """R38-2: evidence that the sealed study has begun (any phase's output or log, or any verdict file). Empty means
    never started: a missing prerequisite is then only "not ready", never drift."""
    evidence = []
    for phase in PHASES:
        evidence += phase_started(phase)
    evidence += [f"verdict {p.name}" for p in (GATE0_PASSED, GATE0_FAILED, SMOKE_PASSED, SMOKE_FAILED) if p.exists()]
    return evidence


def write_receipt(path, lines: list[str]) -> bool:
    """Exclusive write of a stop receipt; False if it already exists (the stop is already latched)."""
    try:
        with open(path, "x", encoding="utf-8", newline="\n") as handle:
            handle.write("\n".join(lines) + "\n")
        return True
    except FileExistsError:
        return False


def record_input_drift(phase: str, where: str, problems: list[str]) -> None:
    """N3: observed input drift is a durable global stop, not a transient deferral."""
    stamp = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
    write_receipt(DRIFT_STOP, [f"# INPUT DRIFT STOP written {stamp} by run_stage1b-v5.py ({phase}, {where})",
                               "Frozen inputs changed while Stage 1b was running; every later run and verdict is "
                               "refused. Owner review required (R28 N3)."] + [f"- {p}" for p in problems[:50]])

def free_mib() -> int:
    return shutil.disk_usage("C:\\").free // (1024 * 1024)


def ac_only() -> bool:
    """Strict local AC predicate: ignores the inherited battery override (run_pilot.py:73-81)."""
    status = run_pilot.power_status()
    return status is not None and status[0] == 1


def wait_for_power() -> bool:
    """True only on AC power with STOP absent at return; STOP is checked on every pass (R24:124)."""
    deadline = time.time() + POWER_WAIT_S
    while True:
        if STOP_FILE.exists():
            return False
        if ac_only():
            return not STOP_FILE.exists()
        if time.time() > deadline:
            return False
        time.sleep(30)


def frozen_paths(phase: str | None = None) -> list[pathlib.Path]:
    paths = [pathlib.Path(p) for p in freeze_targets()] + [SEAL, FREEZE, RUNNER_MANIFEST]
    if phase == "main":
        paths += analysis_paths()                         # R41-1
    for tree_id, root, excludes in EXTERNAL_TREES:
        paths += tree_files(root, excludes)
    if RUNNER.is_dir():
        paths += tree_files(RUNNER)
    return paths


def stat_snapshot(paths) -> dict:
    snapshot = {}
    for path in paths:
        try:
            info = os.stat(path)
            snapshot[str(path)] = (info.st_size, info.st_mtime_ns)
        except OSError:
            snapshot[str(path)] = None
    return snapshot


def stat_drift(snapshot: dict) -> list[str]:
    """Metadata check between full verifications: size/mtime of every frozen file and tree file, added tree files,
    required-absent paths. It detects ordinary edits; it does not authenticate bytes (R28 N3)."""
    current = stat_snapshot(pathlib.Path(p) for p in snapshot)
    drift = [path for path, value in snapshot.items() if current.get(path) != value]
    drift += [pathlib.Path(p).as_posix() + " (now exists)" for p in ABSENT_REQUIRED if pathlib.Path(p).exists()]
    for tree_id, root, excludes in EXTERNAL_TREES:      # a file added to an external tree after the snapshot
        if pathlib.Path(root).is_dir():
            added = [str(p) for p in tree_files(root, excludes) if str(p) not in snapshot]
            drift += [f"{p} (added)" for p in added[:3]]
    return drift


class Admission:
    """F5 + R28 N3/N5 + R38-3: one lock; STOP, latches (incl. the drift stop), frozen-input stat drift, disk, AC power
    and then STOP and the latches AGAIN are checked immediately before launch, and 410 MiB is reserved per running
    job. Observed drift writes the durable INPUT-DRIFT-STOP.txt."""

    def __init__(self, phase: str, snapshot: dict | None):
        self.phase, self.snapshot = phase, snapshot
        self.lock = threading.Lock()
        self.reserved = 0
        self.halted = None

    def admit(self, name: str) -> str | None:
        with self.lock:
            if self.halted:
                return f"deferred-halted ({self.halted})"
            if STOP_FILE.exists():
                return "deferred-stop"
            latch = latch_problems(self.phase)
            if latch:
                self.halted = latch[0]
                return "deferred-latch"
            drift = stat_drift(self.snapshot) if self.snapshot is not None else []
            if drift:
                self.halted = f"frozen input drift: {drift[0]}"
                record_input_drift(self.phase, f"before job {name}", drift)
                return "stopped-input-drift"
            free = free_mib()
            if free - FLOOR_MIB - self.reserved < NEED_MIB:
                return f"deferred-disk ({free} MiB free, {self.reserved} MiB reserved)"
            if not ac_only():                    # N5: after the queue and the long stat scan, before the launch
                return "deferred-on-battery (AC lost before launch)"
            if STOP_FILE.exists():               # R38-3: STOP may arrive during the long scan / disk / AC work
                return "deferred-stop (STOP arrived during the admission checks)"
            latch = latch_problems(self.phase)   # R38-3: so may a failure verdict or the drift stop
            if latch:
                self.halted = latch[0]
                return "deferred-latch (latched during the admission checks)"
            self.reserved += NEED_MIB            # nothing slow runs between these rechecks and the reservation
            return None

    def release(self) -> None:
        with self.lock:
            self.reserved -= NEED_MIB


ADMISSION: Admission | None = None
_INSTANCE_HANDLE = None


def _os_lock(handle) -> None:
    """Non-blocking exclusive lock of the first byte; the OS releases it when the process ends."""
    import msvcrt
    handle.seek(0)
    msvcrt.locking(handle.fileno(), msvcrt.LK_NBLCK, 1)


def acquire_instance_lock(opener=open, locker=None) -> None:
    """N5: reservations are per process, so only one `run` may be active at a time (any phase)."""
    global _INSTANCE_HANDLE
    locker = _os_lock if locker is None else locker
    handle = opener(INSTANCE_LOCK, "a+b")
    try:
        locker(handle)
    except OSError as error:
        handle.close()
        raise SystemExit(f"refused: another run_stage1b-v5.py instance holds {INSTANCE_LOCK.name} ({error})")
    _INSTANCE_HANDLE = handle


def run_job(phase: str, name: str, arm: str, cell: str, mode: str, label: str):
    if state_of(phase, name) != "pending":
        return name, state_of(phase, name)
    log = LOG_ROOT / phase / f"{name}.log"
    if log.exists():
        return name, "log-exists-pending"
    if STOP_FILE.exists():
        return name, "deferred-stop"
    if latch_problems(phase):
        return name, "deferred-latch"
    if free_mib() - FLOOR_MIB < NEED_MIB:
        return name, f"deferred-disk ({free_mib()} MiB free)"
    if not wait_for_power():
        return name, "deferred-stop" if STOP_FILE.exists() else "deferred-on-battery"
    admission = ADMISSION if ADMISSION is not None else Admission(phase, None)
    refusal = admission.admit(name)          # immediately before launch (R24:130; R28 N5)
    if refusal:
        return name, refusal
    try:
        env = dict(os.environ, PYTHONDONTWRITEBYTECODE="1", PYTHONIOENCODING="utf-8",
                   RB_WORLD_CONFIG=str(run_t38.world_file(cell)))
        env.pop("RB_FORECAST_CONFIG", None)
        started = time.time()
        completed = subprocess.run(command(phase, name, arm, cell, mode, label), cwd=str(run_pilot.REPO),
                                   check=False, capture_output=True, text=True, encoding="utf-8", errors="replace",
                                   env=env)
        with open(log, "x", encoding="utf-8", newline="\n") as handle:
            anchor = f" analysis-binding {ANALYSIS_BINDING_SHA}" if phase == "main" else ""   # R41-1
            handle.write(f"returncode={completed.returncode} seconds={time.time() - started:.0f}{anchor}\n"
                         + completed.stdout + completed.stderr)
    finally:
        admission.release()
    return name, state_of(phase, name)


def post_phase_problems(phase: str) -> list[str]:
    """N3: after every phase (main included) the full frozen-input verification runs again (v4: plus the analysis
    freeze for main; v5: against the analysis binding and the identity held since the start); drift writes the
    durable stop receipt."""
    problems, _ = input_integrity_problems(phase)
    problems += analysis_problems(phase)
    if problems:
        record_input_drift(phase, "after the phase", problems)
    return problems


def finish_phase(phase: str) -> list[str]:
    """R38-2: the guaranteed end of every attempted phase (normal, partial, deferred or after a worker / launch
    exception): release keep-awake, then the full post-phase verification. Never raises, so the caller's original
    exception is preserved; a verification that cannot run is itself a persisted failure."""
    try:
        run_pilot.keep_awake(False)
    except Exception as error:
        print(f"keep-awake release failed ({type(error).__name__}: {error})", flush=True)
    try:
        post = post_phase_problems(phase)
    except Exception as error:
        post = [f"post-phase verification could not run ({type(error).__name__}: {error})"]
        try:
            record_input_drift(phase, "after the phase (verification could not run)", post)
        except Exception as receipt_error:
            post.append(f"{DRIFT_STOP.name} could not be written ({type(receipt_error).__name__}: {receipt_error})")
    if post:
        print(f"POST-PHASE INPUT DRIFT ({len(post)} problem(s)); {DRIFT_STOP.name} written; Stage 1b stops:", flush=True)
        for problem in post[:20]:
            print("  -", problem, flush=True)
    return post


def write_freeze() -> int:
    problems = config_problems() + cell_problems(sorted(set(TEST_CELLS) | {SMOKE_CELL})) + reference_problems()
    problems += [f"refused: {p.name} exists (freeze before any verdict)" for p in
                 (GATE0_PASSED, GATE0_FAILED, SMOKE_PASSED, SMOKE_FAILED, DRIFT_STOP) if p.exists()]
    problems += [f"required-absent path exists: {p}" for p in ABSENT_REQUIRED if pathlib.Path(p).exists()]
    problems += forbidden_env()
    roots = [pathlib.Path(root).as_posix() for _, root, _ in EXTERNAL_TREES]
    tools = {name: tool_resolution(name, "job") for name in REQUIRED_TOOLS}
    for name, path in tools.items():                         # R38 s6: both contexts, frozen trees or exception
        if path is None:
            problems.append(f"tool {name} does not resolve")
        elif tool_resolution(name, "driver") != path:
            problems.append(f"tool {name} resolves differently for the driver: {tool_resolution(name, 'driver')}")
        else:
            problems += tool_location_problems(name, path, roots)          # R41-4: by path, not by name
    problems += pth_audit_problems(roots)
    if problems:
        raise SystemExit(f"freeze refused, input problems: {problems}")
    stamp = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
    tree_lines = []
    for tree_id, root, excludes in EXTERNAL_TREES:
        lines = tree_body(root, excludes)
        text = ce.tree_text(lines)
        with open(external_manifest(tree_id), "x", encoding="utf-8", newline="\n") as handle:
            handle.write(text)
        tree_lines.append(f"tree {tree_id} {text.split()[2]} {len(lines)} {pathlib.Path(root).as_posix()}")
    lines = [f"# stage1b input freeze v5, written {stamp}; SHA-256 and absolute path of every input the jobs and "
             f"the comparator read, the external execution trees, the resolved tools, the steering environment and "
             f"required-absent paths"]
    for target in freeze_targets():
        lines.append(f"{sha256_file(target)}  {pathlib.Path(target).as_posix()}")
    lines += tree_lines
    lines += [f"tool {name} {sha256_file(path)} {path}" for name, path in sorted(tools.items())]
    lines += [f"env {name}={value}" for name, value in sorted(frozen_env().items())]
    lines += [f"absent {pathlib.Path(p).as_posix()}" for p in ABSENT_REQUIRED]
    with open(FREEZE, "x", encoding="utf-8", newline="\n") as handle:
        handle.write("\n".join(lines) + "\n")
    print(f"wrote {FREEZE.name}: {len(freeze_targets())} files, {len(tree_lines)} trees, {len(tools)} tools")
    return 0

            print("example env: RB_WORLD_CONFIG=" + str(run_t38.world_file(items[0][2])), "cwd", run_pilot.REPO)
        return 0
    if mode == "freeze":
        return write_freeze()
    acquire_instance_lock()
    global ANALYSIS_BINDING_SHA
    evidence = _guard("study state", study_started)      # an unreadable state counts as started (conservative)
    integrity, identity = input_integrity_problems(phase)
    lost = list(integrity) if evidence else []           # R38-2: never started -> not ready; started -> loss
    analysis = analysis_problems(phase)
    if analysis and (phase_started(phase) or (phase == "main" and ANALYSIS_BINDING.exists())):
        lost += analysis                                 # R41-1: a bound analysis that changed is drift
    if lost:
        record_input_drift(phase, f"at start-up; the study had started ({evidence[0] if evidence else phase})", lost)
        raise SystemExit(f"input integrity lost after the study started; {DRIFT_STOP.name} written: {lost}")
    problems = integrity + gating_problems(phase, identity) + analysis
    if problems:
        raise SystemExit(f"input problems: {problems}")
    if phase == "main":                                  # R41-1: retain the analysis identity before any main job
        if not ANALYSIS_BINDING.exists():
            ANALYSIS_BINDING_SHA = bind_analysis()
        else:
            ANALYSIS_BINDING_SHA = sha256_file(ANALYSIS_BINDING)
    ADMISSION = Admission(phase, stat_snapshot(frozen_paths(phase)))
    (OUT_ROOT / phase).mkdir(parents=True, exist_ok=True)
    (LOG_ROOT / phase).mkdir(parents=True, exist_ok=True)
    run_pilot.keep_awake(True)
    started, deferred, post = time.time(), 0, []
    try:
        with concurrent.futures.ThreadPoolExecutor(run_pilot.WORKERS) as pool:
            futures = [pool.submit(run_job, phase, *item) for item in items]
            try:
                for done, future in enumerate(concurrent.futures.as_completed(futures), start=1):
                    name, label = future.result()
                    deferred += label.startswith(("deferred", "stopped")) or label == "log-exists-pending"
                    print(f"[{time.time() - started:7.0f}s] {name:<34} {label:<24} ({done}/{len(items)})", flush=True)
            except BaseException:
                for future in futures:                   # v4: no further launch after a worker / launch error
                    future.cancel()
                raise
    finally:                                             # R38-2: after attempted work in every case; the original
        post = finish_phase(phase)                       # exception (if any) propagates unchanged
    return 4 if post else (3 if deferred else 0)
